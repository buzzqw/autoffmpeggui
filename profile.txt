# AutoFFmpegGui - practical FFmpeg delivery and archive presets
# Format:  Name;ffmpeg video options
# CRF quality is applied by the GUI quality slider, so do NOT set -crf here
# (except for fixed presets like lossless).
# The GUI adds the selected audio, subtitle, scaling and container options.
# Profiles below target common playback devices, not only scene releases.

# --- x264 8-bit (scene / P2P) ---
x264 1080p (Scene);-c:v libx264 -preset veryslow -profile:v high -x264-params ref=16:bframes=16:b-adapt=2:direct=auto:me=umh:subme=10:rc-lookahead=60:trellis=2:aq-mode=2:keyint=250:min-keyint=25:scenecut=40:weightp=2
x264 1080p (Web-DL);-c:v libx264 -preset slow -profile:v high -x264-params ref=8:bframes=8:b-adapt=2:direct=auto:me=hex:subme=9:rc-lookahead=40:trellis=1:aq-mode=2
x264 720p (Scene);-c:v libx264 -preset veryslow -profile:v high -x264-params ref=9:bframes=16:b-adapt=2:direct=auto:me=umh:subme=10:rc-lookahead=60:trellis=2:aq-mode=2:keyint=250:min-keyint=25:scenecut=40
x264 576p/SD (Scene);-c:v libx264 -preset veryslow -profile:v high -x264-params ref=5:bframes=16:b-adapt=2:direct=auto:me=umh:subme=10:rc-lookahead=60:trellis=2:aq-mode=2

# --- HEVC / x265 (release groups) ---
HEVC 10-bit (x265 QxR/PSA);-c:v libx265 -preset slow -pix_fmt yuv420p10le -x265-params ref=5:bframes=6:b-adapt=2:rc-lookahead=60:scenecut=40:keyint=240:min-keyint=24:me=star:subme=5
HEVC 10-bit (x265 FGT/D-Z0N3);-c:v libx265 -preset slow -pix_fmt yuv420p10le -x265-params ref=4:bframes=4:b-adapt=2:rc-lookahead=60:scenecut=40:keyint=250:min-keyint=25:me=umh:subme=4
HEVC 8-bit 1080p (x265);-c:v libx265 -preset slow -pix_fmt yuv420p -x265-params ref=4:bframes=4:rc-lookahead=40:scenecut=40

# --- AV1 / VP9 (web streaming) ---
AV1 SVT (Web 8-bit);-c:v libsvtav1 -preset 8 -pix_fmt yuv420p
AV1 SVT (Web 10-bit);-c:v libsvtav1 -preset 8 -pix_fmt yuv420p10le
VP9 (Web);-c:v libvpx-vp9 -deadline good -cpu-used 4 -row-mt 1

# --- Xvid / MPEG-4 (legacy) ---
Xvid (Scene-style);-c:v libxvid -vtag XVID
MPEG-4 Good;-c:v mpeg4

# --- Fixed presets ---
x264 Lossless (archive);-c:v libx264 -preset slow -crf 0

# --- TV / living-room compatibility ---
TV H.264 1080p (USB);-c:v libx264 -preset medium -profile:v high -level:v 4.1 -pix_fmt yuv420p -movflags +faststart
TV H.264 720p (USB);-c:v libx264 -preset medium -profile:v high -level:v 3.1 -pix_fmt yuv420p -movflags +faststart
TV HEVC 4K 10-bit (HDR);-c:v libx265 -preset medium -pix_fmt yuv420p10le -tag:v hvc1
TV MPEG-2 (legacy);-c:v mpeg2video -pix_fmt yuv420p -qmin 2 -qmax 12

# --- Phones / tablets ---
Phone H.264 1080p (wide compatibility);-c:v libx264 -preset fast -profile:v main -level:v 4.1 -pix_fmt yuv420p -movflags +faststart
Phone H.264 720p (small);-c:v libx264 -preset fast -profile:v main -level:v 3.1 -pix_fmt yuv420p -movflags +faststart
Phone HEVC 10-bit (modern);-c:v libx265 -preset medium -pix_fmt yuv420p10le -tag:v hvc1 -movflags +faststart

# --- Web / streaming ---
Web H.264 (YouTube/Vimeo);-c:v libx264 -preset slow -profile:v high -pix_fmt yuv420p -movflags +faststart
Web AV1 (quality);-c:v libsvtav1 -preset 6 -pix_fmt yuv420p -svtav1-params tune=0
Web VP9 (quality);-c:v libvpx-vp9 -deadline good -cpu-used 2 -row-mt 1 -tile-columns 2 -frame-parallel 1

# --- Editing / mezzanine ---
Editing ProRes 422;-c:v prores_ks -profile:v 2 -pix_fmt yuv422p10le
Editing ProRes 4444;-c:v prores_ks -profile:v 4 -pix_fmt yuv444p10le
Editing DNxHR HQX;-c:v dnxhd -profile:v dnxhr_hqx -pix_fmt yuv422p10le

# --- High quality archive ---
Archive FFV1 lossless;-c:v ffv1 -level 3 -coder 1 -context 1 -g 1 -slicecrc 1
Archive H.265 10-bit;-c:v libx265 -preset slower -pix_fmt yuv420p10le
Archive AV1 10-bit;-c:v libsvtav1 -preset 4 -pix_fmt yuv420p10le -svtav1-params tune=0
